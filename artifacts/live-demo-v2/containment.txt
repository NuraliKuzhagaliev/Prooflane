import json
from decimal import Decimal

PATH = "/tmp/input.csv"
EXPECTED = ["invoice_id","vendor","quantity","unit_price","contract_price","notes"]

def parse_price(s):
    d = Decimal(s)
    if d < 0 or d.as_tuple().exponent < -2:
        raise ValueError("bad price")
    return d

def fmt(d):
    return format(d.quantize(Decimal("0.01")), "f")

with open(PATH, encoding="utf-8-sig", newline="") as f:
    reader = csv.DictReader(f)
    if reader.fieldnames != EXPECTED:
        raise ValueError("bad header")
    rows = list(reader)

if not (1 <= len(rows) <= 500):
    raise ValueError("bad row count")

total_billed = Decimal(0)
total_variance = Decimal(0)
findings = []

for i, r in enumerate(rows, start=2):
    q = int(r["quantity"])
    if q <= 0:
        raise ValueError("bad quantity")
    up = parse_price(r["unit_price"])
    cp = parse_price(r["contract_price"])
    total_billed += q * up
    diff = up - cp
    if diff > 0:
        v = q * diff
        total_variance += v
        findings.append({"row": i, "variance": fmt(v)})

print(json.dumps({
    "row_count": len(rows),
    "total_billed": fmt(total_billed),
    "total_variance": fmt(total_variance),
    "findings": findings,
}))
